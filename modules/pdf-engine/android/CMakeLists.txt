cmake_minimum_required(VERSION 3.22.1)
project(versara_pdf_editor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PDFIUM_ARCH_arm64-v8a arm64)
set(PDFIUM_ARCH_armeabi-v7a arm)
set(PDFIUM_ARCH_x86 x86)
set(PDFIUM_ARCH_x86_64 x64)
set(PDFIUM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/android-${PDFIUM_ARCH_${ANDROID_ABI}}")
if(NOT EXISTS "${PDFIUM_DIR}/lib/libpdfium.so")
  message(FATAL_ERROR "Run npm run native:pdfium before compiling the native PDF editor.")
endif()
add_library(pdfium SHARED IMPORTED)
set_target_properties(pdfium PROPERTIES IMPORTED_LOCATION "${PDFIUM_DIR}/lib/libpdfium.so")
add_library(versara_pdf_editor SHARED ../cpp/TextEditor.cpp src/main/cpp/TextEditorJni.cpp)
target_include_directories(versara_pdf_editor PRIVATE "${PDFIUM_DIR}/include" ../cpp)
target_link_libraries(versara_pdf_editor pdfium)
target_link_options(versara_pdf_editor PRIVATE "-Wl,-z,max-page-size=16384")
